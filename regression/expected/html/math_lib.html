$$\begin{array}{l} \textbf{function } square(x) \\ \left| \begin{array}{l} \quad \text{return } x^{2} \end{array} \right. \end{array}$$
$$\begin{array}{l} \textbf{function } cube(x) \\ \left| \begin{array}{l} \quad \text{return } x^{3} \end{array} \right. \end{array}$$
$$\begin{array}{l} \textbf{function } factorial(n) \\ \left| \begin{array}{l} \quad \textbf{if } n \leq 1 \textbf{ then} \\ \quad \left| \begin{array}{l} \quad \text{return } 1 \end{array} \right.\\ \quad \textbf{else} \\ \quad \left| \begin{array}{l} \quad \text{return } n \cdot factorial(n - 1) \end{array} \right. \end{array} \right. \end{array}$$
$$\begin{array}{l} \textbf{function } pythagoras(a, b) \\ \left| \begin{array}{l} \quad \text{return } \sin(0) + a^{2} + b^{2} \end{array} \right. \end{array}$$
